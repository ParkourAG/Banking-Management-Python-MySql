/////////// Create Bank Database ///////////
create database bank_database;

/////////// Account Details Database ///////////
CREATE TABLE accounts_details (
    id INT AUTO_INCREMENT PRIMARY KEY,
    acc_name VARCHAR(30) NOT NULL,
    ph_no VARCHAR(15) NOT NULL,
    email VARCHAR(100) NOT NULL,
    money DECIMAL(10,2) DEFAULT 0.00,
    create_date DATE NOT NULL DEFAULT (CURRENT_DATE),
    create_time TIME NOT NULL DEFAULT (CURRENT_TIME),
    acc_status ENUM('active', 'inactive') NOT NULL DEFAULT 'active'
);

/////////// Transactions Database ///////////
CREATE TABLE transactions(
transaction_id int auto_increment primary key,
id INT NOT NULL ,
money DECIMAL(10,2) NOT NULL,
tran_type ENUM('deposit', 'withdraw') NOT NULL,
tran_date date default (current_date),
tran_time time default (current_time)
)AUTO_INCREMENT = 1000;

/////////// Employees Database ///////////
CREATE TABLE employees(
    emp_id INT AUTO_INCREMENT PRIMARY KEY,
    acc_name VARCHAR(30) NOT NULL,
    ph_no VARCHAR(15) NOT NULL,
    email VARCHAR(100) NOT NULL,
    join_date DATE NOT NULL DEFAULT (CURRENT_DATE),
    leave_date TIME NOT NULL DEFAULT (CURRENT_TIME),
    admin_position ENUM('po', 'officer') NOT NULL
);
